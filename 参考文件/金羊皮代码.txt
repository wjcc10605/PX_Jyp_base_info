--金羊皮正式版订单转化数据
SELECT  
a.下单时间 时间,
a.游戏名称,
a.分月付下单数,
a.分月付下单支付数,
a.分月付下单完单数,
b.分月付完单数

FROM    (
            SELECT  UNIX_TIMESTAMP(TO_DATE(create_time)) * 1000 下单时间
                    -- TO_CHAR(create_time,'yyyymmdd') 下单时间
                    ,game_name 游戏名称
                    ,COUNT(DISTINCT order_item_id) 分月付下单数
                    ,COUNT(DISTINCT CASE    WHEN receipt_status_code = 3 THEN order_item_id END) 分月付下单支付数
                    ,COUNT(DISTINCT CASE    WHEN order_item_status_code = 4 THEN order_item_id END) 分月付下单完单数
            FROM    yj_new_dw.dwd_pxb7_trade_order_item_df
            WHERE   ds = MAX_PT('yj_new_dw.dwd_pxb7_trade_order_item_df')
            AND     finance_order_flag_code = 1
            AND     TO_CHAR(create_time,'yyyymmdd') >= '20251219'
            AND     TO_CHAR(create_time,'yyyymmdd') <= to_char(date_add(GETDATE(), -1),'yyyymmdd')
            AND     game_name NOT IN ('bahei盒子')
            GROUP BY 下单时间,游戏名称
        ) a
LEFT JOIN   (
                SELECT  UNIX_TIMESTAMP(TO_DATE(complete_time)) * 1000 完单时间
                        -- TO_CHAR(complete_time,'yyyymmdd') 完单时间
                        ,game_name 游戏名称
                        ,COUNT(DISTINCT CASE    WHEN order_item_status_code = 4 THEN order_item_id END) 分月付完单数
                FROM    yj_new_dw.dwd_pxb7_trade_order_item_df
                WHERE   ds = MAX_PT('yj_new_dw.dwd_pxb7_trade_order_item_df')
                AND     finance_order_flag_code = 1
                AND     order_item_status_code = 4
                AND     TO_CHAR(complete_time,'yyyymmdd') >= '20251219'
                AND     TO_CHAR(complete_time,'yyyymmdd') <= to_char(date_add(GETDATE(), -1),'yyyymmdd')
                AND     game_name NOT IN ('bahei盒子')
                GROUP BY 完单时间,游戏名称
            ) b
ON      a.下单时间 = b.完单时间
AND     a.游戏名称 = b.游戏名称

GROUP BY a.下单时间,a.游戏名称,a.分月付下单数,a.分月付下单支付数,a.分月付下单完单数,b.分月付完单数
;

select game_name,sum(product_list)
,sum(is_detail_from_list)
,sum(high)
,sum(is_order)
,sum(is_pay)
,sum(is_complete)
from 
(

select t.game_name
        ,merge_id
        ,case when event_key = 'productOnshow'  and page_name  in  ('商品列表页面','搜索结果页')  then 1 else 0 end product_list
        ,case when event_key = 'productDetailPageView'  then 1   else 0 end detail  
         ,IF(t.event_key = 'productDetailPageView' 
         AND t.business_type <> '卖家'
                    AND     GET_JSON_OBJECT(json_attributes,'$.referrerPageName_var') IN ('商品列表页面','搜索结果页'),1,0) AS is_detail_from_list
        ,case when event_key = 'productDetailClick' AND page_name = '商品详情页面' and pit_name in ('咨询','立即购买','还价') then 1 else 0 end   high
        ,case when b.buyer_id is not null  then 1 else 0 end is_order
        ,case when b.buyer_id is not null and order_item_status NOT in('待支付','已取消') then 1 else 0 end is_pay
        ,case when b.buyer_id is not null and order_item_status_code=4 then 1 else 0 end is_complete
from yj_new_dw.dws_gio_event_di t 
left join yj_new_dw.dws_pxb7_product_info_df d on d.product_id = t.product_id and d.ds = 20251228
left join yj_new_dw.ods_pxb7_product_product_promotion_price_snapshot_df c on d.product_id = c.product_id and c.ds = 20251228
left join yj_new_dw.dws_pxb7_trade_order_item_summary_df b on b.ds = 20251228 and t.product_id = b.product_id 
where t.ds BETWEEN 20251224 AND 20251228
and d.price/100 >=500
and t.data_source_id in ('PC','H5')
AND     t.event_key NOT IN ('$ma_channel_touch','sellerContractSigned','buyerContractSigned')
AND     NOT (
            t.event_key = 'flowOnshow'
            AND     pit_name = '返回顶部'
) 
--and b.order_item_status_code = 4
and b.service_mode_code = 1
and to_char(b.order_time,'yyyymmdd') between 20251224 and 20251228
--and c.product_id is not null 
--AND c.biz_type = 'finance' AND c.extend IS NOT NULL AND c.extend<>''
AND (biz_type <> 'finance' or extend IS  NULL OR extend='')
)
group by game_name
;
